import numpy as np

def manual_convolution(x, h):
    # Length of the sequences
    len_x = len(x)
    len_h = len(h)

    # Length of the output sequence
    len_y = len_x + len_h - 1

    # Initialize the output sequence with zeros
    y = [0] * len_y

    # Perform convolution
    for n in range(len_y):
        for k in range(len_x):
            if 0 <= n - k < len_h:
                y[n] += x[k] * h[n - k]

    return y

# Define the input sequences
x = [1, -1, -2, 3, -1]
h = [1, 2, 3]

# Built-in function
numpy_convolution = np.convolve(x, h)

# Our created function
manual_result = manual_convolution(x, h)

# Print the input sequences
print("Input sequence x:", x)
print("Input sequence h:", h)

# Print the convolution results
print("Convolution result using NumPy:", numpy_convolution)
print("Convolution result using manual calculation:", manual_result)

circular Convolution
# Defining sequences
a = [1, -1, -2, 3, -1]
b = [1, 2, 3, 0]

len_x = len(a)
len_y = len(b)
n = max(len_x, len_y)  # Using max length for circular convolution

# Zero-padding to match lengths
a += [0] * (n - len_x)
b += [0] * (n - len_y)

# Output sequence
c = [0] * n

print("Input sequence a: ", a)
print("Input sequence b: ", b)

for i in range(n):
    for j in range(n):
        c[i] += a[j] * b[(i - j) % n]

print("Circular Convoluted Sequence: ", c)

import cmath
import numpy as np
import matplotlib.pyplot as plt

# Compute the Discrete Fourier Transform (DFT)
def optimized_dft(X):
    N = len(X)
    X_dft = []
    for K in range(N):
        sum_val = 0
        for n in range(N):
            angle = -2j * cmath.pi * K * n / N
            sum_val += X[n] * cmath.exp(angle)
        X_dft.append(sum_val)
    return X_dft

# Input Sequence
X = [1, -1, 2, -2, 3, -3, 4, -4, -5]
print("Input Sequence:", X)

# Compute DFT using our Custom method

import cmath
import matplotlib.pyplot as plt
import numpy as np

# Compute the Inverse Discrete Fourier Transform (IDFT)
def optimized_idft(x):
    N = len(x)
    # Generate the conjugate twiddle factors for IDFT
    w_matrix = np.array([[cmath.exp(-2j * cmath.pi * k * m / N) for m in range(N)] for k in range(N)])
    W = np.linalg.inv(w_matrix)
    X = []
    for k in range(N):
        sum_val = 0
        for n in range(N):
            sum_val += x[n] * W[k, n]
        X.append(sum_val)
    return X

# Input sequence (DFT coefficients)
X = [5, 0, complex(1, -1), 0, 1, 0, complex(1, 1), 0]

# Compute IDFT using custom method
x_optimized = optimized_idft(X)

# Compute IDFT using NumPy's ifft
x_ifft = np.fft.ifft(X)

# Print results
print("From Our Method (IDFT):")
for i, value in enumerate(x_optimized):
    print(f"x[{i}] = {value.real:.3f} + {value.imag:.3f}j")

print("\nFrom NumPy IFFT:")
for i, value in enumerate(x_ifft):
    print(f"x[{i}] = {value.real:.3f} + {value.imag:.3f}j")

# Plot magnitude of IDFT (time-domain signal)

plt.tight_layout()

import numpy as np

def overlap_save(x, h, L):
    x = np.asarray(x)
    h = np.asarray(h)
    M = len(h)
    N = L + M - 1
    if N < M:
        raise ValueError("Block size N (L + M - 1) must be at least as long as the filter h.")

    x_padded = np.concatenate((np.zeros(M - 1), x))
    desired_length = len(x) + M - 1
    num_blocks = int(np.ceil(desired_length / L))
    total_length_required = (num_blocks - 1) * L + N

    if len(x_padded) < total_length_required:
        x_padded = np.concatenate((x_padded, np.zeros(total_length_required - len(x_padded))))

    H_fft = np.fft.fft(h, n=N)
    y_segments = []

    for m in range(num_blocks):
        start = m * L
        end = start + N
        x_m = x_padded[start:end]
        X_fft = np.fft.fft(x_m, n=N)
        Y_fft = X_fft * H_fft
        y_m = np.fft.ifft(Y_fft).real
        y_m_valid = y_m[M - 1:]
        y_segments.append(y_m_valid)

    y = np.concatenate(y_segments)
    return y[:desired_length]

def main():
    x = [3, -1, 0, 1, 3, 2, 0, 1, 2, 1]
    h = [1, 1, 1]

    print("Input signal x(n) =", x)
    print("Impulse response h(n) =", h)

    try:
        L = int(input(f"Enter L (number of valid new samples per block): "))
    except ValueError:
        L = 3
    if L <= 0:
        L = 3

    y_overlap = overlap_save(x, h, L)
    y_direct = np.convolve(x, h)

    print("\nDirect convolution result y(n) using np.convolve =")
    print(y_direct)
    print("\nFinal overlap-save convolution result y(n) =")
    print(np.round(y_overlap, 3))

if __name__ == '__main__':
    main()

import numpy as np

def overlap_add(x, h, L, decimals=3):
    x = np.asarray(x)
    h = np.asarray(h)
    M = len(h)
    N = L + M - 1

    H = np.fft.fft(h, N)
    num_blocks = int(np.ceil(len(x) / L))
    x_padded = np.pad(x, (0, num_blocks * L - len(x)), mode='constant') # x is padded
    y = np.zeros(len(x_padded) + M - 1)

    for m in range(num_blocks):
        x_m = x_padded[m * L : (m + 1) * L] # x_padded is divided into smaller blocks
        x_m_padded = np.pad(x_m, (0, N - L), mode='constant')
        Y_m = np.fft.fft(x_m_padded) * H
        y_m = np.fft.ifft(Y_m).real
        y[m * L : m * L + N] += y_m

    y = y[:len(x) + M - 1]
    return np.round(y, decimals)

# Input values
x = [3, -1, 0, 1, 3, 2, 0, 1, 2, 1]
h = [1, 1, 1]
L = 3
decimals = 3

y_overlap_add = overlap_add(x, h, L, decimals)
y_direct = np.round(np.convolve(x, h), decimals)

print("Direct convolution:", y_direct)
print("Overlap-add convolution:", y_overlap_add)

import numpy as np

def bit_reverse_indices(N):
    bits = int(np.log2(N))
    return [int(f'{i:0{bits}b}'[::-1], 2) for i in range(N)]

def fft_radix2_dit(x):
    N = len(x)
    x = np.array(x, dtype=complex)

    # Bit-reverse input
    rev_indices = bit_reverse_indices(N)
    x = x[rev_indices]

    # Iterative FFT
    stages = int(np.log2(N))
    for s in range(1, stages + 1):
        m = 2 ** s
        wm = np.exp(-2j * np.pi / m)
        for k in range(0, N, m):
            w = 1
            for j in range(m // 2):
                t = w * x[k + j + m//2]
                u = x[k + j]
                x[k + j] = u + t
                x[k + j + m//2] = u - t
                w *= wm
    return x

# Input sequence
x = [1, 2, 3, 3, 2, 1, 0, 1]
X = fft_radix2_dit(x)

# Output
for i, val in enumerate(X):
    print(f"X[{i}] = {val:.4f}")

import numpy as np

def bit_reverse(x):
    N = len(x)
    bits = int(np.log2(N))
    indices = [int(f'{i:0{bits}b}'[::-1], 2) for i in range(N)]
    return x[indices]

def fft_radix2_dif(x):
    x = np.array(x, dtype=complex)
    N = len(x)
    for s in range(int(np.log2(N)), 0, -1):
        m = 2 ** s
        wm = np.exp(-2j * np.pi / m)
        for k in range(0, N, m):
            w = 1
            for j in range(m // 2):
                u, t = x[k + j], x[k + j + m//2]
                x[k + j] = u + t
                x[k + j + m//2] = (u - t) * w
                w *= wm
    return bit_reverse(x)

# Example usage
x = [1, 2, 3, 3, 2, 1, 0, 1]
X = fft_radix2_dif(x)
for i, val in enumerate(X):
    print(f"X[{i}] = {val:.3f}")
