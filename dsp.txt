************************************************************************************************************************************************
******************************************************************** Overlap Save **************************************************************

import numpy as np

def overlap_save(x, h, L):
    x = np.asarray(x)
    h = np.asarray(h)
    M = len(h)
    N = L + M - 1
    if N < M:
        raise ValueError("Block size N (L + M - 1) must be at least as long as the filter h.")

    x_padded = np.concatenate((np.zeros(M - 1), x))
    desired_length = len(x) + M - 1
    num_blocks = int(np.ceil(desired_length / L))
    total_length_required = (num_blocks - 1) * L + N

    if len(x_padded) < total_length_required:
        x_padded = np.concatenate((x_padded, np.zeros(total_length_required - len(x_padded))))

    H_fft = np.fft.fft(h, n=N)
    y_segments = []

    for m in range(num_blocks):
        start = m * L
        end = start + N
        x_m = x_padded[start:end]
        X_fft = np.fft.fft(x_m, n=N)
        Y_fft = X_fft * H_fft
        y_m = np.fft.ifft(Y_fft).real
        y_m_valid = y_m[M - 1:]
        y_segments.append(y_m_valid)

    y = np.concatenate(y_segments)
    return y[:desired_length]

def main():
    x = [3, -1, 0, 1, 3, 2, 0, 1, 2, 1]
    h = [1, 1, 1]

    print("Input signal x(n) =", x)
    print("Impulse response h(n) =", h)

    try:
        L = int(input(f"Enter L (number of valid new samples per block): "))
    except ValueError:
        L = 3
    if L <= 0:
        L = 3

    y_overlap = overlap_save(x, h, L)
    y_direct = np.convolve(x, h)

    print("\nDirect convolution result y(n) using np.convolve =")
    print(y_direct)
    print("\nFinal overlap-save convolution result y(n) =")
    print(np.round(y_overlap, 3))

if __name__ == '__main__':
    main()

******************************************************************** Overlap Add ***************************************************************

import numpy as np

def overlap_add(x, h, L, decimals=3):
    x = np.asarray(x)
    h = np.asarray(h)
    M = len(h)
    N = L + M - 1

    H = np.fft.fft(h, N)
    num_blocks = int(np.ceil(len(x) / L))
    x_padded = np.pad(x, (0, num_blocks * L - len(x)), mode='constant') # x is padded
    y = np.zeros(len(x_padded) + M - 1)

    for m in range(num_blocks):
        x_m = x_padded[m * L : (m + 1) * L] # x_padded is divided into smaller blocks
        x_m_padded = np.pad(x_m, (0, N - L), mode='constant')
        Y_m = np.fft.fft(x_m_padded) * H
        y_m = np.fft.ifft(Y_m).real
        y[m * L : m * L + N] += y_m

    y = y[:len(x) + M - 1]
    return np.round(y, decimals)

# Input values
x = [3, -1, 0, 1, 3, 2, 0, 1, 2, 1]
h = [1, 1, 1]
L = 3
decimals = 3

y_overlap_add = overlap_add(x, h, L, decimals)
y_direct = np.round(np.convolve(x, h), decimals)

print("Direct convolution:", y_direct)
print("Overlap-add convolution:", y_overlap_add)

******************************************************************** DIT FFT *******************************************************************

import numpy as np

def bit_reverse_indices(N):
    bits = int(np.log2(N))
    return [int(f'{i:0{bits}b}'[::-1], 2) for i in range(N)]

def fft_radix2_dit(x):
    N = len(x)
    x = np.array(x, dtype=complex)

    # Bit-reverse input
    rev_indices = bit_reverse_indices(N)
    x = x[rev_indices]

    # Iterative FFT
    stages = int(np.log2(N))
    for s in range(1, stages + 1):
        m = 2 ** s
        wm = np.exp(-2j * np.pi / m)
        for k in range(0, N, m):
            w = 1
            for j in range(m // 2):
                t = w * x[k + j + m//2]
                u = x[k + j]
                x[k + j] = u + t
                x[k + j + m//2] = u - t
                w *= wm
    return x

# Input sequence
x = [1, 2, 3, 3, 2, 1, 0, 1]
X = fft_radix2_dit(x)

# Output
for i, val in enumerate(X):
    print(f"X[{i}] = {val:.4f}")

************************************************************************************************************************************************
******************************************************************** DIF FFT *******************************************************************

import numpy as np

def bit_reverse(x):
    N = len(x)
    bits = int(np.log2(N))
    indices = [int(f'{i:0{bits}b}'[::-1], 2) for i in range(N)]
    return x[indices]

def fft_radix2_dif(x):
    x = np.array(x, dtype=complex)
    N = len(x)
    for s in range(int(np.log2(N)), 0, -1):
        m = 2 ** s
        wm = np.exp(-2j * np.pi / m)
        for k in range(0, N, m):
            w = 1
            for j in range(m // 2):
                u, t = x[k + j], x[k + j + m//2]
                x[k + j] = u + t
                x[k + j + m//2] = (u - t) * w
                w *= wm
    return bit_reverse(x)

# Example usage
x = [1, 2, 3, 3, 2, 1, 0, 1]
X = fft_radix2_dif(x)
for i, val in enumerate(X):
    print(f"X[{i}] = {val:.3f}")
